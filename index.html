Honor Pad X9a — Termux Bağlantı Rehberi

Bu rehber her zaman elinin altında. Köprüyü başlattıktan sonra Ayarlar'a Köprü anahtarını girip kaydetmen yeterli.

1. Depolama ve paketleri kur (Termux) 
termux-setup-storage
pkg update -y && pkg install -y python nodejs termux-api
termux-setup-storage tabletteki ~/storage/shared (İndirilenler, Belgeler, DCIM) klasörlerine erişim açar.
2. Yerel köprü sunucusunu oluştur (~/jarvis_bridge.py) 
cat > ~/jarvis_bridge.py << 'EOF'
import json, os, subprocess
from http.server import BaseHTTPRequestHandler, ThreadingHTTPServer

TOKEN = os.environ.get("JARVIS_TOKEN", "degistir-bunu")
HOST, PORT = "127.0.0.1", 8765

def exec_tool(name, args):
    if name == "get_status":
        p = subprocess.run(["termux-battery-status"], capture_output=True, text=True)
        return p.stdout or "Pil bilgisi alınamadı."
    if name == "list_files":
        target = os.path.expanduser(args.get("path", "~/storage/shared"))
        return "\n".join(sorted(os.listdir(target)))
    if name == "read_file":
        target = os.path.expanduser(args["path"])
        with open(target, "r", encoding="utf-8", errors="replace") as f:
            return f.read()[:12000]
    if name == "write_file":
        target = os.path.expanduser(args["path"])
        os.makedirs(os.path.dirname(target) or ".", exist_ok=True)
        with open(target, "w", encoding="utf-8") as f:
            f.write(args["content"])
        return "Yazıldı: " + target
    if name == "run_shell":
        p = subprocess.run(args["command"], shell=True, capture_output=True, text=True, timeout=25)
        return (p.stdout + p.stderr).strip()[:8000] or "Tamamlandı."
    if name == "open_app":
        subprocess.Popen(["am", "start", "-a", "android.intent.action.MAIN", "-c", "android.intent.category.LAUNCHER", args["package"]], stdout=subprocess.DEVNULL, stderr=subprocess.DEVNULL)
        return "Uygulama başlatıldı: " + args["package"]
    if name == "open_url":
        subprocess.run(["termux-open-url", args["url"]], check=False)
        return "Açıldı: " + args["url"]
    if name == "notify":
        subprocess.run(["termux-notification", "-t", args.get("title", "Jarvis"), "-c", args["text"]], check=False)
        return "Bildirim gönderildi."
    if name == "clipboard_get":
        p = subprocess.run(["termux-clipboard-get"], capture_output=True, text=True)
        return p.stdout
    if name == "clipboard_set":
        subprocess.run(["termux-clipboard-set", args["text"]], check=False)
        return "Panoya kopyalandı."
    if name == "tts_speak":
        subprocess.run(["termux-tts-speak", "-l", "tr-TR", args["text"]], check=False)
        return "Okundu."
    return "Bilinmeyen araç"

class Handler(BaseHTTPRequestHandler):
    def _cors(self):
        self.send_header("Access-Control-Allow-Origin", "*")
        self.send_header("Access-Control-Allow-Headers", "Content-Type, Authorization")
        self.send_header("Access-Control-Allow-Methods", "POST, GET, OPTIONS")
    def do_OPTIONS(self):
        self.send_response(204); self._cors(); self.end_headers()
    def do_GET(self):
        self.send_response(200); self._cors(); self.send_header("Content-Type", "application/json"); self.end_headers()
        self.wfile.write(b'{"ok":true,"device":"Honor Pad X9a"}')
    def do_POST(self):
        if self.headers.get("Authorization") != "Bearer " + TOKEN:
            self.send_response(401); self._cors(); self.end_headers(); self.wfile.write(b'{"error":"Unauthorized"}'); return
        length = int(self.headers.get("Content-Length", 0))
        data = json.loads(self.rfile.read(length) or b"{}")
        try:
            res = exec_tool(data.get("name"), data.get("args") or {})
            self.send_response(200); self._cors(); self.send_header("Content-Type", "application/json"); self.end_headers()
            self.wfile.write(json.dumps({"result": res}, ensure_ascii=False).encode())
        except Exception as e:
            self.send_response(500); self._cors(); self.send_header("Content-Type", "application/json"); self.end_headers()
            self.wfile.write(json.dumps({"error": str(e)}, ensure_ascii=False).encode())
    def log_message(self, *a): pass

print(f"Jarvis köprü aktif -> http://{HOST}:{PORT} | Token: {TOKEN}")
ThreadingHTTPServer((HOST, PORT), Handler).serve_forever()
EOF
3. Her seferinde tek komutla başlat (Kısayol: jstart) 
echo 'export JARVIS_TOKEN="jarvis-x9a-2026"; termux-wake-lock; python ~/jarvis_bridge.py' >> ~/.bashrc
source ~/.bashrc
Termux'u açtığında JARVIS_TOKEN değeri olan jarvis-x9a-2026 anahtarını Jarvis Ayarlar'a yaz. termux-wake-lock ekran uyusa bile köprünün açık kalmasına yardım eder.
4. Sistem geneli / ADB yetkileri (İsteğe bağlı)
Honor Pad X9a'da normal Android uygulamalarının ötesinde kabuk/sistem ayarlarına erişmek istersen Geliştirici Seçenekleri > Kablosuz Hata Ayıklama (veya Shizuku) üzerinden adb connect ile Termux içinden adb shell komutları çalıştırabilirsin.
